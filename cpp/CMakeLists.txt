cmake_minimum_required(VERSION 3.15)
project(raim_core LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(pybind11 CONFIG REQUIRED)
pybind11_add_module(raim_core src/bindings.cpp)
if(MSVC)
  target_compile_options(raim_core PRIVATE /O2 /W3)
else()
  target_compile_options(raim_core PRIVATE -O2 -Wall)
endif()
install(TARGETS raim_core LIBRARY DESTINATION .)
